-- Gym Attendance and Membership Management System
-- Initial foundation migration
-- Timezone: Asia/Manila | Currency: PHP
--
-- IMPORTANT:
-- This migration is a foundation, not a complete production schema.
-- Do not deploy until reviewed and tested against PostgreSQL/Supabase.

begin;

create extension if not exists pgcrypto with schema extensions;

create schema if not exists private;

-- Prevent public/anonymous access to internal application tables.
revoke all on schema private from public;
revoke all on schema private from anon, authenticated;

-- ============================================================
-- 1. Gym configuration
-- ============================================================

create table public.gym_settings (
    setting_key text primary key,
    setting_value jsonb not null,
    updated_at timestamptz not null default now(),
    constraint gym_settings_key_not_blank
        check (length(trim(setting_key)) > 0)
);

insert into public.gym_settings (
    setting_key,
    setting_value
)
values
    ('timezone', '"Asia/Manila"'::jsonb),
    ('currency', '"PHP"'::jsonb)
on conflict (setting_key) do nothing;

-- ============================================================
-- 2. Owner access registry
-- ============================================================
-- Owner access is granted only to a verified Supabase Auth user.
-- This table does not create an Owner or enable public registration.
-- Owner bootstrap will be implemented separately.

create table private.owner_access (
    user_id uuid primary key references auth.users(id) on delete restrict,
    is_active boolean not null default false,
    created_at timestamptz not null default now(),
    deactivated_at timestamptz,
    constraint owner_access_deactivation_consistent
        check (
            (is_active and deactivated_at is null)
            or
            (not is_active)
        )
);

-- ============================================================
-- 3. Member records
-- ============================================================
-- Do not store member PINs or plaintext PIN credentials here.
-- PIN enrollment and secure identifier lookup are future migrations.


create table private.members (
    id uuid primary key default gen_random_uuid(),
    member_code text not null unique,
    full_name text not null,
    mobile_lookup_hash bytea not null unique,
    mobile_encrypted bytea not null,
    status text not null default 'pending',
    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now(),
    approved_at timestamptz,

    constraint members_code_not_blank
        check (length(trim(member_code)) > 0),

    constraint members_name_not_blank
        check (length(trim(full_name)) > 0),

    constraint members_mobile_lookup_hash_not_empty
        check (octet_length(mobile_lookup_hash) > 0),

    constraint members_mobile_encrypted_not_empty
        check (octet_length(mobile_encrypted) > 0),

    constraint members_status_valid
        check (status in ('pending', 'active', 'suspended'))
);


create index members_status_idx
    on private.members (status);

-- ============================================================
-- 4. Audit event foundation
-- ============================================================
-- Never place PINs, passwords, full payment credentials, or other
-- secrets in event details. Append-only enforcement and hash-chain
-- verification will be added before production use.

create table private.audit_events (
    id bigint generated always as identity primary key,
    actor_user_id uuid references auth.users(id) on delete restrict,
    event_type text not null,
    subject_type text not null,
    subject_id text,
    details jsonb not null default '{}'::jsonb,
    created_at timestamptz not null default now(),
    constraint audit_event_type_not_blank
        check (length(trim(event_type)) > 0),
    constraint audit_subject_type_not_blank
        check (length(trim(subject_type)) > 0)
);

create index audit_events_created_at_idx
    on private.audit_events (created_at desc);

create index audit_events_actor_created_idx
    on private.audit_events (actor_user_id, created_at desc);

-- ============================================================
-- 5. Default access protections
-- ============================================================

alter table public.gym_settings enable row level security;
alter table private.owner_access enable row level security;
alter table private.members enable row level security;
alter table private.audit_events enable row level security;

-- No direct client table access. Later migrations will expose only
-- narrowly authorized functions and views after their policies
-- and execution permissions have been designed and tested.

revoke all on public.gym_settings from public, anon, authenticated;
revoke all on private.owner_access from public, anon, authenticated;
revoke all on private.members from public, anon, authenticated;
revoke all on private.audit_events from public, anon, authenticated;

commit;

-- Membership core: plans and one-month purchases.
-- Draft only: do not deploy before PostgreSQL testing.

begin;

create table private.membership_plans (
    id uuid primary key default gen_random_uuid(),
    name text not null,
    price_php numeric(12,2) not null,
    is_active boolean not null default true,
    created_at timestamptz not null default now(),

    constraint membership_plan_name_not_blank
        check (length(trim(name)) > 0),

    constraint membership_plan_price_positive
        check (price_php > 0)
);

create table private.membership_purchases (
    id uuid primary key default gen_random_uuid(),
    member_id uuid not null
        references private.members(id) on delete restrict,
    plan_id uuid not null
        references private.membership_plans(id) on delete restrict,

    -- Snapshot the agreed price so future plan edits do not
    -- change the historical purchase price.
    price_php numeric(12,2) not null,

    purchase_status text not null default 'pending_payment',

    -- Dates use the gym's local calendar, not the server's timezone.
    activation_date date,
    expiry_date date,

    created_at timestamptz not null default now(),

    constraint membership_purchase_price_positive
        check (price_php > 0),

    constraint membership_purchase_status_valid
        check (
            purchase_status in (
                'pending_payment',
                'pending_activation',
                'active',
                'expired',
                'payment_review'
            )
        ),

    constraint membership_purchase_dates_consistent
        check (
            (activation_date is null and expiry_date is null)
            or
            (
                activation_date is not null
                and expiry_date is not null
                and expiry_date >= activation_date
            )
        )
);

create index membership_purchases_member_idx
    on private.membership_purchases (member_id, created_at desc);

create index membership_purchases_plan_idx
    on private.membership_purchases (plan_id);

alter table private.membership_plans enable row level security;
alter table private.membership_purchases enable row level security;

revoke all on private.membership_plans
    from public, anon, authenticated;

revoke all on private.membership_purchases
    from public, anon, authenticated;

commit;

-- Payment ledger foundation.
-- Draft only: not production-ready until constraints,
-- permissions, and transactional operations are tested.

begin;

-- Actual money received. This record is separate from what
-- purchase or walk-in request the receipt is allocated to.
create table private.payment_records (
    id uuid primary key default gen_random_uuid(),

    amount_php numeric(12,2) not null,
    payment_method text not null,
    received_at timestamptz not null default now(),

    verification_status text not null default 'recorded',

    external_reference text,
    evidence_reference text,

    recorded_by uuid references auth.users(id) on delete restrict,
    created_at timestamptz not null default now(),

    constraint payment_amount_positive
        check (amount_php > 0),

    constraint payment_method_valid
        check (
            payment_method in (
                'cash',
                'gcash',
                'bank_transfer'
            )
        ),

    constraint payment_verification_status_valid
        check (
            verification_status in (
                'recorded',
                'confirmed',
                'rejected',
                'confirmation_voided_in_error'
            )
        )
);

-- Whole-receipt allocation. A receipt may target one membership
-- purchase or one walk-in request, never both.
create table private.payment_allocations (
    id uuid primary key default gen_random_uuid(),

    payment_id uuid not null unique
        references private.payment_records(id) on delete restrict,

    membership_purchase_id uuid
        references private.membership_purchases(id) on delete restrict,

    walkin_request_id uuid,

    allocated_by uuid references auth.users(id) on delete restrict,
    allocated_at timestamptz not null default now(),

    constraint payment_allocation_exactly_one_target
        check (
            (membership_purchase_id is not null)
            <>
            (walkin_request_id is not null)
        )
);

create index payment_allocations_membership_idx
    on private.payment_allocations (membership_purchase_id)
    where membership_purchase_id is not null;

create index payment_allocations_walkin_idx
    on private.payment_allocations (walkin_request_id)
    where walkin_request_id is not null;

-- Append-only event history. Enforcement of append-only behavior
-- and authorization will be added in the security/operations phase.
create table private.payment_events (
    id bigint generated always as identity primary key,

    payment_id uuid not null
        references private.payment_records(id) on delete restrict,

    event_type text not null,
    reason text,
    actor_user_id uuid references auth.users(id) on delete restrict,
    created_at timestamptz not null default now(),

    constraint payment_event_type_not_blank
        check (length(trim(event_type)) > 0)
);

create index payment_events_payment_created_idx
    on private.payment_events (payment_id, created_at);

-- Correction proposals preserve the original receipt. They do not
-- themselves modify the payment or authorize access.
create table private.payment_corrections (
    id uuid primary key default gen_random_uuid(),

    payment_id uuid not null
        references private.payment_records(id) on delete restrict,

    proposed_amount_php numeric(12,2),
    proposed_payment_method text,
    reason text not null,
    evidence_reference text,

    correction_status text not null default 'proposed',

    proposed_by uuid references auth.users(id) on delete restrict,
    proposed_at timestamptz not null default now(),

    decided_by uuid references auth.users(id) on delete restrict,
    decided_at timestamptz,
    decision_reason text,

    constraint payment_correction_reason_not_blank
        check (length(trim(reason)) > 0),

    constraint payment_correction_amount_valid
        check (
            proposed_amount_php is null
            or proposed_amount_php > 0
        ),

    constraint payment_correction_method_valid
        check (
            proposed_payment_method is null
            or proposed_payment_method in (
                'cash',
                'gcash',
                'bank_transfer'
            )
        ),

    constraint payment_correction_status_valid
        check (
            correction_status in (
                'proposed',
                'approved',
                'rejected',
                'withdrawn'
            )
        )
);

create index payment_corrections_payment_idx
    on private.payment_corrections (payment_id, proposed_at desc);

-- Protect the tables from direct browser access.
alter table private.payment_records enable row level security;
alter table private.payment_allocations enable row level security;
alter table private.payment_events enable row level security;
alter table private.payment_corrections enable row level security;

revoke all on private.payment_records
    from public, anon, authenticated;

revoke all on private.payment_allocations
    from public, anon, authenticated;

revoke all on private.payment_events
    from public, anon, authenticated;

revoke all on private.payment_corrections
    from public, anon, authenticated;

commit;

-- Walk-in and attendance foundation.
-- Draft only: do not deploy before database validation.

begin;

-- Versioned walk-in fees preserve the amount applicable when
-- a request is created. Configure actual fees before accepting visits.
create table private.walkin_fee_versions (
    id uuid primary key default gen_random_uuid(),
    fee_php numeric(12,2) not null,
    currency text not null default 'PHP',
    effective_from timestamptz not null default now(),
    created_by uuid references auth.users(id) on delete restrict,
    created_at timestamptz not null default now(),

    constraint walkin_fee_positive check (fee_php > 0),
    constraint walkin_fee_currency_php check (currency = 'PHP')
);

-- A public request is provisional; it is not proof of payment
-- or a valid visit. Only trusted server-side operations should
-- confirm payment and finalize attendance.
create table private.walkin_requests (
    id uuid primary key default gen_random_uuid(),
    fee_version_id uuid not null
        references private.walkin_fee_versions(id) on delete restrict,

    fee_php numeric(12,2) not null,
    currency text not null default 'PHP',

    request_status text not null default 'pending',
    created_at timestamptz not null default now(),
    expires_at timestamptz not null,

    confirmed_by uuid references auth.users(id) on delete restrict,
    confirmed_at timestamptz,

    constraint walkin_request_fee_positive check (fee_php > 0),
    constraint walkin_request_currency_php check (currency = 'PHP'),
    constraint walkin_request_expiry_after_creation
        check (expires_at > created_at),
    constraint walkin_request_status_valid
        check (
            request_status in (
                'pending',
                'confirmed',
                'expired',
                'payment_review',
                'rejected'
            )
        ),
    constraint walkin_confirmation_consistent
        check (
            (request_status = 'confirmed'
                and confirmed_by is not null
                and confirmed_at is not null)
            or
            (request_status <> 'confirmed')
        )
);

-- Attendance is check-in only. There is no checkout or duration field.
create table private.attendance_checkins (
    id uuid primary key default gen_random_uuid(),

    member_id uuid
        references private.members(id) on delete restrict,

    walkin_request_id uuid
        references private.walkin_requests(id) on delete restrict,

    gym_local_date date not null,
    checked_in_at timestamptz not null default now(),

    attendance_status text not null default 'valid',

    constraint attendance_exactly_one_subject
        check (
            (member_id is not null)
            <>
            (walkin_request_id is not null)
        ),

    constraint attendance_status_valid
        check (
            attendance_status in (
                'valid',
                'invalidated_payment',
                'invalidated_correction',
                'review'
            )
        )
);

-- Only one member check-in per gym-local calendar day.
create unique index attendance_one_member_valid_checkin_per_day
    on private.attendance_checkins (member_id, gym_local_date)
    where member_id is not null and attendance_status = 'valid';

-- Each walk-in request may have at most one attendance record.
create unique index attendance_one_checkin_per_walkin
    on private.attendance_checkins (walkin_request_id)
    where walkin_request_id is not null;

create index attendance_date_idx
    on private.attendance_checkins (gym_local_date desc);

create index attendance_member_date_idx
    on private.attendance_checkins (member_id, gym_local_date desc)
    where member_id is not null;

-- Record why a previous attendance record became invalid.
-- Original attendance history remains available for review.
create table private.attendance_validity_resolutions (
    id uuid primary key default gen_random_uuid(),

    attendance_id uuid not null
        references private.attendance_checkins(id) on delete restrict,

    prior_status text not null,
    new_status text not null,
    reason text not null,

    resolved_by uuid references auth.users(id) on delete restrict,
    resolved_at timestamptz not null default now(),

    constraint attendance_resolution_reason_not_blank
        check (length(trim(reason)) > 0),

    constraint attendance_resolution_status_valid
        check (
            prior_status in (
                'valid',
                'invalidated_payment',
                'invalidated_correction',
                'review'
            )
            and new_status in (
                'valid',
                'invalidated_payment',
                'invalidated_correction',
                'review'
            )
        )
);

-- Protect all new tables from direct browser access.
alter table private.walkin_fee_versions enable row level security;
alter table private.walkin_requests enable row level security;
alter table private.attendance_checkins enable row level security;
alter table private.attendance_validity_resolutions enable row level security;

revoke all on private.walkin_fee_versions
    from public, anon, authenticated;

revoke all on private.walkin_requests
    from public, anon, authenticated;

revoke all on private.attendance_checkins
    from public, anon, authenticated;

revoke all on private.attendance_validity_resolutions
    from public, anon, authenticated;

commit;

-- Member PIN security foundation.
-- Draft only. Do not enable public PIN operations until a trusted
-- server-side verifier and rate limits are implemented and tested.

begin;

create table private.member_pin_credentials (
    member_id uuid primary key
        references private.members(id) on delete restrict,

    -- Store only an encoded Argon2id verifier, never the PIN itself.
    -- The trusted backend must generate and validate this value.
    password_hash text not null,

    credential_version integer not null default 1,
    created_at timestamptz not null default now(),
    updated_at timestamptz not null default now(),

    constraint member_pin_hash_not_blank
        check (length(trim(password_hash)) > 0),

    constraint member_pin_version_positive
        check (credential_version > 0)
);

-- Hashed, short-lived, single-use enrollment/reset capabilities.
-- Raw tokens must only be returned once to the intended recipient.
create table private.member_pin_setup_tokens (
    id uuid primary key default gen_random_uuid(),

    member_id uuid not null
        references private.members(id) on delete restrict,

    token_hash text not null unique,
    token_purpose text not null,
    expires_at timestamptz not null,

    consumed_at timestamptz,
    created_at timestamptz not null default now(),

    constraint pin_token_hash_not_blank
        check (length(trim(token_hash)) > 0),

    constraint pin_token_purpose_valid
        check (token_purpose in ('enrollment', 'reset')),

    constraint pin_token_expiry_after_creation
        check (expires_at > created_at)
);

create index member_pin_setup_tokens_member_idx
    on private.member_pin_setup_tokens (member_id, created_at desc);

create index member_pin_setup_tokens_expiry_idx
    on private.member_pin_setup_tokens (expires_at)
    where consumed_at is null;

-- Keyed identifiers only. The actual HMAC keys and rate-limit
-- policy must be handled by a trusted server-side component.
create table private.pin_attempt_buckets (
    id uuid primary key default gen_random_uuid(),

    bucket_type text not null,
    identifier_digest text not null,
    window_started_at timestamptz not null,
    failed_attempts integer not null default 0,
    blocked_until timestamptz,

    updated_at timestamptz not null default now(),

    constraint pin_bucket_type_valid
        check (
            bucket_type in ('member_identifier', 'trusted_network')
        ),

    constraint pin_identifier_digest_not_blank
        check (length(trim(identifier_digest)) > 0),

    constraint pin_failed_attempts_nonnegative
        check (failed_attempts >= 0),

    constraint pin_bucket_unique_window
        unique (bucket_type, identifier_digest, window_started_at)
);

create index pin_attempt_buckets_blocked_idx
    on private.pin_attempt_buckets (blocked_until)
    where blocked_until is not null;

-- No direct access from browser-facing roles.
alter table private.member_pin_credentials enable row level security;
alter table private.member_pin_setup_tokens enable row level security;
alter table private.pin_attempt_buckets enable row level security;

    from public, anon, authenticated;

    from public, anon, authenticated;

-- Cash reconciliation foundation.
-- Draft only: validate transactions and permissions before deployment.

begin;

-- One drawer session represents one physical cash drawer.
create table private.cash_drawer_sessions (
    id uuid primary key default gen_random_uuid(),

    opened_at timestamptz not null default now(),
    opened_by uuid not null
        references auth.users(id) on delete restrict,

    opening_float_php numeric(12,2) not null default 0,

    session_status text not null default 'open',

    closed_at timestamptz,
    closed_by uuid references auth.users(id) on delete restrict,

    constraint cash_opening_float_nonnegative
        check (opening_float_php >= 0),

    constraint cash_drawer_status_valid
        check (session_status in ('open', 'closed')),

    constraint cash_drawer_close_consistent
        check (
            (session_status = 'open'
                and closed_at is null
                and closed_by is null)
            or
            (session_status = 'closed'
                and closed_at is not null
                and closed_by is not null)
        )
);

-- At most one open drawer session.
create unique index cash_drawer_only_one_open_session
    on private.cash_drawer_sessions ((session_status))
    where session_status = 'open';

-- A reconciliation is a snapshot, not a mutable running total.
create table private.cash_reconciliations (
    id uuid primary key default gen_random_uuid(),

    drawer_session_id uuid not null unique
        references private.cash_drawer_sessions(id) on delete restrict,

    expected_cash_php numeric(12,2) not null,
    counted_cash_php numeric(12,2) not null,
    variance_php numeric(12,2)
        generated always as (counted_cash_php - expected_cash_php) stored,

    reconciled_by uuid not null
        references auth.users(id) on delete restrict,

    reconciled_at timestamptz not null default now(),
    notes text,

    constraint reconciliation_expected_cash_nonnegative
        check (expected_cash_php >= 0),

    constraint reconciliation_counted_cash_nonnegative
        check (counted_cash_php >= 0)
);

-- Adjustments must be explained and separately recorded.
-- This table does not itself change a historical snapshot.
create table private.cash_reconciliation_adjustments (
    id uuid primary key default gen_random_uuid(),

    reconciliation_id uuid not null
        references private.cash_reconciliations(id) on delete restrict,

    amount_delta_php numeric(12,2) not null,
    reason text not null,
    evidence_reference text,

    recorded_by uuid not null
        references auth.users(id) on delete restrict,

    created_at timestamptz not null default now(),

    constraint cash_adjustment_nonzero
        check (amount_delta_php <> 0),

    constraint cash_adjustment_reason_not_blank
        check (length(trim(reason)) > 0)
);

create index cash_reconciliations_date_idx
    on private.cash_reconciliations (reconciled_at desc);

create index cash_adjustments_reconciliation_idx
    on private.cash_reconciliation_adjustments
       (reconciliation_id, created_at);

alter table private.cash_drawer_sessions enable row level security;
alter table private.cash_reconciliations enable row level security;
alter table private.cash_reconciliation_adjustments enable row level security;

revoke all on private.cash_drawer_sessions
    from public, anon, authenticated;

revoke all on private.cash_reconciliations
    from public, anon, authenticated;

revoke all on private.cash_reconciliation_adjustments
    from public, anon, authenticated;

commit;

-- Audit integrity foundation.
-- Draft only: validate hash-chain behavior and transaction handling
-- in a disposable PostgreSQL/Supabase environment before deployment.

begin;

-- Each stream has its own chain head so unrelated audit streams
-- can be serialized independently.
create table private.audit_chain_heads (
    stream_key text primary key,
    last_event_id bigint,
    last_event_hash text,
    updated_at timestamptz not null default now(),

    constraint audit_chain_stream_not_blank
        check (length(trim(stream_key)) > 0),

    constraint audit_chain_head_consistent
        check (
            (last_event_id is null and last_event_hash is null)
            or
            (last_event_id is not null and last_event_hash is not null)
        )
);

-- Add hash-chain fields to the foundation audit table.
alter table private.audit_events
    add column stream_key text,

begin;

-- The API schema is exposed by local/hosted Supabase configuration.
create schema if not exists api;

-- Check a user ID only through a server-side call after the Edge Function
-- has verified the user's access token with Supabase Auth.
create or replace function api.is_active_owner_for_user(p_user_id uuid)
returns boolean
language sql
stable
security definer
set search_path = ''
as $function$
    select exists (
        select 1
        from private.owner_access oa
        where oa.user_id = p_user_id
          and oa.is_active = true
    );
$function$;

-- Do not permit browser roles to call this function.
revoke all on schema api from public, anon, authenticated;
revoke all on function api.is_active_owner_for_user(uuid)
    from public, anon, authenticated, service_role;

-- Only the server role can invoke this check.
grant usage on schema api to service_role;
grant execute on function api.is_active_owner_for_user(uuid)
    to service_role;

commit;
